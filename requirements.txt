# ====================
# RASPBERRY PI 5 INSTALLATION NOTES
# ====================
# 
# STEP 1: System packages (run these first!)
#   sudo apt update
#   sudo apt install -y python3-pip python3-venv
#   sudo apt install -y python3-numpy python3-pil python3-smbus
#   sudo apt install -y python3-picamera2 libcamera-apps
#   sudo apt install -y i2c-tools
#
# STEP 2: Enable interfaces
#   sudo raspi-config
#   - Enable I2C (Interface Options > I2C)
#   - Enable Camera (Interface Options > Legacy Camera - if needed)
#   - Enable Serial (Interface Options > Serial Port)
#
# STEP 3: Create virtual environment with system packages
#   python3 -m venv venv --system-site-packages
#   source venv/bin/activate
#
# STEP 4: Install Python packages
#   pip install -r requirements.txt
#
# STEP 5: Verify I2C devices
#   i2cdetect -y 1
#   (Should show addresses: 0x27=LCD, 0x29=TCS34725, 0x68=MPU6050)
#
# STEP 6: Test camera
#   libcamera-hello
